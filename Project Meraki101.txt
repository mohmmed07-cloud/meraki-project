const Navbar = () => {

import React, { useState, useEffect } from 'react';
import { db, auth } from './firebase';
import { collection, addDoc } from 'firebase/firestore';
import { onAuthStateChanged } from 'firebase/auth';
import { useNavigate } from 'react-router-dom';

const AddMotionAdmin = () => {
    const [title, setTitle] = useState('');
    const [description, setDescription] = useState('');
    const [image1File, setImage1File] = useState(null);
    const [image2File, setImage2File] = useState(null);
    const [image3File, setImage3File] = useState(null);
    const [loading, setLoading] = useState(false);
    const [successMsg, setSuccessMsg] = useState('');
    const [checkingAuth, setCheckingAuth] = useState(true);
    
    const navigate = useNavigate();

    useEffect(() => {
        const unsubscribe = onAuthStateChanged(auth, (user) => {
            if (!user) {
                navigate('/login');
            } else {
                setCheckingAuth(false);
            }
        });
        return () => unsubscribe();
    }, [navigate]);

    // دالة لتحويل ملف الصورة من الهاتف إلى صيغة نصية (Base64) لتخزينها في قاعدة البيانات وعرضها فوراً
    const convertFileToBase64 = (file) => {
        return new Promise((resolve, reject) => {
            if (!file) {
                resolve(null);
                return;
            }
            const reader = new FileReader();
            reader.onload = (e) => resolve(e.target.result);
            reader.onerror = (error) => reject(error);
            reader.readAsDataURL(file);
        });
    };

    const handleSubmit = async (e) => {
        e.preventDefault();
        setLoading(true);
        setSuccessMsg('');

        try {
            // تحويل الصور الثلاث المأخوذة من الهاتف إلى نصوص برمجية
            const img1Base64 = await convertFileToBase64(image1File);
            const img2Base64 = await convertFileToBase64(image2File);
            const img3Base64 = await convertFileToBase64(image3File);

            // تجميع الصور الموجودة فقط في مصفوفة السلايدر
            const imagesArray = [img1Base64, img2Base64, img3Base64].filter(img => img !== null);

            // حفظ البيانات في الـ Firestore
            await addDoc(collection(db, "motions"), {
                title,
                description,
                images: imagesArray,
                createdAt: new Date()
            });

            setSuccessMsg('تم نشر الموشن وصور الهاتف بنجاح! 🚀');
            setTitle('');
            setDescription('');
            setImage1File(null);
            setImage2File(null);
            setImage3File(null);
        } catch (error) {
            console.error("Error adding motion: ", error);
            alert('حدث خطأ أثناء النشر');
        } finally {
            setLoading(false);
        }
    };

    if (checkingAuth) return <div style={{textAlign: 'center', marginTop: '50px'}}>Loading...</div>;

    return (
        <div style={{padding: '40px', maxWidth: '600px', margin: '0 auto'}}>
            <h2>Admin: Add Motion with Phone Images</h2>
            {successMsg && <p style={{color: 'green'}}>{successMsg}</p>}
            
            <form onSubmit={handleSubmit} style={{display: 'flex', flexDirection: 'column', gap: '15px'}}>
                <div>
                    <label>Motion Title:</label>
                    <input 
                        type="text" 
                        value={title} 
                        onChange={(e) => setTitle(e.target.value)} 
                        style={{width: '100%', padding: '8px'}}
                        required 
                    />
                </div>

                <div>
                    <label>Description:</label>
                    <textarea 
                        value={description} 
                        onChange={(e) => setDescription(e.target.value)} 
                        rows="4" 
                        style={{width: '100%', padding: '8px'}}
                        required 
                    />
                </div>

                <div>
                    <label>الصورة الأولى (السلايدر اليسار):</label>
                    <input 
                        type="file" 
                        accept="image/*" 
                        onChange={(e) => setImage1File(e.target.files[0])} 
                        style={{width: '100%', padding: '8px'}}
                    />
                </div>

                <div>
                    <label>الصورة الثانية (الرئيسية في الوسط):</label>
                    <input 
                        type="file" 
                        accept="image/*" 
                        onChange={(e) => setImage2File(e.target.files[0])} 
                        style={{width: '100%', padding: '8px'}}
                    />
                </div>

                <div>
                    <label>الصورة الثالثة (السلايدر اليمين):</label>
                    <input 
                        type="file" 
                        accept="image/*" 
                        onChange={(e) => setImage3File(e.target.files[0])} 
                        style={{width: '100%', padding: '8px'}}
                    />
                </div>

                <button type="submit" disabled={loading} style={{padding: '10px', background: '#1a2a42', color: 'white', border: 'none', cursor: 'pointer', fontWeight: 'bold'}}>
                    {loading ? 'Publishing...' : 'Publish Motion'}
                </button>
            </form>
        </div>
    );
};

export default AddMotionAdmin;

import React, { useState, useEffect } from 'react';
import { collection, getDocs } from "firebase/firestore";
import { db, auth } from './firebase';
import { signOut, onAuthStateChanged } from 'firebase/auth';
import { useNavigate } from 'react-router-dom';
import './Admin.css';

const Admin = () => {
    const [applications, setApplications] = useState([]);
    const [loading, setLoading] = useState(true);
    const [user, setUser] = useState(null);
    const navigate = useNavigate();

    // التحقق مما إذا كان المستخدم مسجلاً لدخوله
    useEffect(() => {
        const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
            if (!currentUser) {
                navigate('/login'); // إذا لم يكن مسجلاً، اذهِب لصفحة تسجيل الدخول
            } else {
                setUser(currentUser);
                fetchApplications();
            }
        });
        return () => unsubscribe();
    }, [navigate]);

    const fetchApplications = async () => {
        try {
            const querySnapshot = await getDocs(collection(db, "applications"));
            const data = querySnapshot.docs.map(doc => ({
                id: doc.id,
                ...doc.data()
            }));
            setApplications(data);
            setLoading(false);
        } catch (error) {
            console.error("Error fetching data: ", error);
            setLoading(false);
        }
    };

    const handleLogout = async () => {
        await signOut(auth);
        navigate('/login');
    };

    if (!user) return null;

    return (
        <div className="admin-container">
    <div className="admin-header-flex">
        <h2>Club Admin Panel - Membership Requests</h2>
        <button className="logout-btn" onClick={handleLogout}>Logout</button>
    </div>
            
            {loading ? (
                <p>Loading data...</p>
            ) : (
                
                <div className="table-wrapper">
 <table className="admin-table">
                        <thead>
                            <tr>
                                <th>Name</th>
                                <th>Age</th>
                                <th>Major</th>
                                <th>Year</th>
                                <th>Email</th>
                                <th>Phone</th>
                                <th>English Level</th>
                            </tr>
                        </thead>
                        <tbody>
                            {applications.map((app) => (
                                <tr key={app.id}>
                                    <td>{app.name}</td>
                                    <td>{app.age}</td>
                                    <td>{app.major}</td>
                                    <td>{app.yearOfStudy}</td>
                                    <td>{app.email}</td>
                                    <td>{app.phone}</td>
                                    <td>{app.englishLevel}</td>
                                </tr>
                            ))}
                        </tbody>
                    </table>
                </div>
            )}
        </div>
    );
};

import React, { useState } from 'react';
import './Applications.css';
import { collection, addDoc } from "firebase/firestore";
import { db } from './firebase';


const Applications = () => {
    // 1. تعريف الحالة لتخزين قيم الحقول
    const [formData, setFormData] = useState({
        name: '',
        age: '',
        major: '',
        email: '',
        phone: '',
        yearOfStudy: '',
        englishLevel: '',
        schoolSystem: '',
        schoolSystemOtherText: '',
        experience: [],
        experienceOtherText: ''
    });
const [statusMessage, setStatusMessage] = useState({ type: '', text: '' });
    // 2. دالة تحديث النصوص وخيارات الراديو
    const handleChange = (e) => {
        setFormData({ ...formData, [e.target.name]: e.target.value });
    };

    // 3. دالة معالجة مربعات الاختيار (Checkboxes)
    const handleCheckboxChange = (e) => {
        const { value, checked } = e.target;
        if (checked) {
            setFormData(prev => ({ ...prev, experience: [...prev.experience, value] }));
        } else {
            setFormData(prev => ({ ...prev, experience: prev.experience.filter(item => item !== value) }));
        }
    };

    // 4. دالة الإرسال إلى Firestore
   const handleSubmit = async (e) => {
  e.preventDefault();
  try {
    const docRef = await addDoc(collection(db, "applications"), formData);
    console.log("Document written with ID: ", docRef.id);

    setStatusMessage({ type: 'success', text: 'تم إرسال طلبك بنجاح ✅' });
    setFormData({
      name: '', age: '', major: '', email: '', phone: '',
      yearOfStudy: '', englishLevel: '', schoolSystem: '',
      schoolSystemOtherText: '', experience: [], experienceOtherText: ''
    });
  } catch (error) {
    console.error("Error adding document: ", error);
    setStatusMessage({ type: 'error', text: 'حدث خطأ أثناء الإرسال، حاول مرة ثانية.' });
  }

  // الرسالة بتختفي لحالها بعد 4 ثواني
  setTimeout(() => setStatusMessage({ type: '', text: '' }), 4000);
};


    return (
        
        <main className="applications-container">
            <h2>Applications</h2>
            {statusMessage.text && (
  <div className={`form-status ${statusMessage.type}`}>
    {statusMessage.text}
  </div>
)}
            <form className="application-form" onSubmit={handleSubmit}>
                
                <label htmlFor="name">Name:</label>
                <input 
                    type="text" 
                    id="name" 
                    name="name" 
                    required 
                    placeholder="Your Name" 
                    value={formData.name} 
                    onChange={handleChange} 
                />    
                
                <label htmlFor="age">Age:</label>
                <input 
                    type="number" 
                    id="age" 
                    name="age" 
                    required 
                    placeholder="Your Age" 
                    value={formData.age} 
                    onChange={handleChange} 
                />
                
                <label htmlFor="major">Major:</label>
                <input 
                    type="text" 
                    id="major" 
                    name="major" 
                    required 
                    placeholder="Your Major" 
                    value={formData.major} 
                    onChange={handleChange} 
                />

                <label htmlFor="email">Email:</label>
                <input 
                    type="email" 
                    id="email" 
                    name="email" 
                    required 
                    placeholder="Your Email@gmail.com" 
                    value={formData.email} 
                    onChange={handleChange} 
                />

                {/* سنة الدراسة */}
                <div className="form-group">
                    <p>Year of study:</p>
                    {['1st year', '2nd year', '3rd year', '4th year', 'Other'].map((year) => (
                        <div key={year} className="radio-option">
                            <input 
                                type="radio" 
                                id={`year-${year}`} 
                                name="yearOfStudy" 
                                value={year} 
                                onChange={handleChange} 
                                checked={formData.yearOfStudy === year} 
                            />
                            <label htmlFor={`year-${year}`}>{year}</label>
                        </div>
                    ))}
                </div>
                 
                <label htmlFor="phone">Phone Number:</label>
                <input 
                    type="tel" 
                    id="phone" 
                    name="phone" 
                    required 
                    placeholder="Your Phone Number" 
                    value={formData.phone} 
                    onChange={handleChange} 
                />
                 
                {/* مستوى اللغة الإنجليزية */}
                <div className="form-group">
                    <p>What english level have you been assigned originally in the level exam?</p>
                    {['A1.1', 'A1.2', 'A2.1', 'A2.2', 'B1.1', 'B1.2', 'B2.1', 'B2.2', 'C1.1', 'C1.2'].map((level) => (
                        <div key={level} className="radio-option">
                            <input 
                                type="radio" 
                                id={`level-${level}`} 
                                name="englishLevel" 
                                value={level} 
                                onChange={handleChange} 
                                checked={formData.englishLevel === level} 
                            />
                            <label htmlFor={`level-${level}`}>{level}</label>
                        </div>
                    ))}
                </div> 

                {/* النظام المدرسي */}
                <div className="form-group">
                    <p>What school system is your school degree based on?</p>
                    {['Jordanian Tawjihi', 'American program', 'IB', 'IGCSE', 'Arabic Degrees'].map((system) => (
                        <div key={system} className="radio-option">
                            <input 
                                type="radio" 
                                id={`system-${system}`} 
                                name="schoolSystem" 
                                value={system} 
                                onChange={handleChange} 
                                checked={formData.schoolSystem === system} 
                            />
                            <label htmlFor={`system-${system}`}>{system}</label>
                        </div>
                    ))}
                    <div className="radio-option other-input" style={{ display: 'flex', alignItems: 'center', gap: '10px' }}>
                        <input 
                            type="radio" 
                            id="system-other" 
                            name="schoolSystem" 
                            value="Other" 
                            onChange={handleChange} 
                            checked={formData.schoolSystem === 'Other'}
                        />
                        <label htmlFor="system-other">Other:</label>
                        <input 
                            type="text" 
                            name="schoolSystemOtherText" 
                            value={formData.schoolSystemOtherText} 
                            onChange={handleChange} 
                            style={{ borderBottom: '1px solid #ccc', borderTop: 'none', borderLeft: 'none', borderRight: 'none', outline: 'none' }} 
                        />
                    </div>
                </div>

                {/* الخبرات السابقة */}
                <div className="form-group">
                    <p>Do you have any experience with debate, MUN, public speaking, presentations?</p>
                    {['Yes, Formal Debate', 'Yes, Public speaking', 'Yes, Presentations', 'Yes, MUN', 'No prior experience'].map((exp) => (
                        <div key={exp} className="checkbox-option">
                            <input 
                                type="checkbox" 
                                id={`exp-${exp}`} 
                                name="experience" 
                                value={exp} 
                                onChange={handleCheckboxChange} 
                                checked={formData.experience.includes(exp)}
                            />
                            <label htmlFor={`exp-${exp}`}>{exp}</label>
                        </div>
                    ))}
                    <div className="checkbox-option other-input" style={{ display: 'flex', alignItems: 'center', gap: '10px' }}>
                        <input 
                            type="checkbox"

                            value="Other"

                            onChange={handleChange}

            </form>
        </main>
    );
};

const ProtectedRoute = ({ children }) => {
    // auth.currentUser يتحقق إذا كان هناك مستخدم مسجل دخول حالياً
    const user = auth.currentUser;

    // إذا لم يكن مسجلاً، قم بتوجيهه لصفحة الـ login
    // ملاحظة: يمكنك أيضاً الاعتماد على onAuthStateChanged إذا احتجت لفحص دقيق للـ session
    if (!user) {
        return <Navigate to="/login" />;
    }
    return children;
};
